# Wishy Fase 1 (PoC) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** PoC web de Wishy: wishlist compartible por link donde los regaladores reservan sin duplicarse y el dueño jamás ve los estados (regla de la sorpresa validada en servidor).

**Architecture:** Next.js App Router full-stack en un repo. Toda la lógica de negocio vive en funciones puras de `src/lib/*` que reciben la conexión `Db` como parámetro — así se testean con PGlite (Postgres in-process) sin docker ni mocks. Las páginas son server components que llaman a esas funciones; las mutaciones son server actions. La regla de la sorpresa se implementa como serialización por allowlist en un único punto de lectura.

**Tech Stack:** Next.js 15 (App Router, TS, Tailwind) · Drizzle ORM · Postgres (Neon en prod, PGlite en tests) · Auth.js v5 con Google · Vitest · Vercel.

**Spec:** `docs/superpowers/specs/2026-08-08-wishy-fase1-design.md`

## Global Constraints

- **Regla de la sorpresa:** el dueño de un item nunca recibe claims, comments, notifications de regaladores ni campos derivados (contadores, disponibilidad). Serialización por **allowlist** (pick explícito de campos), nunca por blocklist/spread.
- Toda validación de la regla es **en servidor**; la UI nunca es la única barrera.
- Copy de la UI en **español rioplatense** (vos/tenés).
- Precio: **texto libre con símbolo** (ej. `"$45.000"`), sin parsing ni multi-moneda.
- Expiración de reservas: **14 días**, sin cron — se expira al leer (lazy).
- Sin email ni push: notificaciones **solo in-app**.
- Mobile-first, base 375px. Acento visual: coral/terracota (`#E2725B`); verde suave para comprado; sin rojo salvo "reserva por vencer".
- Los enums de dominio en la DB en español: `exacto|idea`, `mucho|gustaria|suelta`, `activo|lo_tengo|no_lo_quiero`, `reservado|comprado|liberado|expirado`.
- Commit al final de cada task.

```ts
import { claims } from '@/db/schema'

export async function expireStaleClaims(db: Db, itemId?: string) {
  const vencida = and(eq(claims.estado, 'reservado'), lt(claims.expiresAt, new Date()))
  await db.update(claims)
    .set({ estado: 'expirado', updatedAt: new Date() })
    .where(itemId ? and(vencida, eq(claims.itemId, itemId)) : vencida)
}
```

`src/lib/lists.ts`:

```ts
import { and, eq, inArray } from 'drizzle-orm'
import { claims, comments, guestIdentities, items, lists, users } from '@/db/schema'
import type { Db } from '@/db/types'
import type { Viewer } from '@/lib/types'
import { expireStaleClaims } from '@/lib/claims'
import { pickItemFields, type ClaimView, type GiverItemView, type OwnerItemView } from '@/lib/serialize'

export type ListView =
  | { role: 'owner'; shareToken: string; items: OwnerItemView[] }
  | { role: 'giver'; shareToken: string; ownerNombre: string; items: GiverItemView[] }

export type CommentView = { id: string; autorNombre: string; texto: string; createdAt: Date }

async function loadClaimViews(db: Db, itemIds: string[], viewer: Viewer) {
  if (!itemIds.length) return new Map<string, ClaimView[]>()
  const rows = await db.select({
    id: claims.id, itemId: claims.itemId, estado: claims.estado,
    notaQueCompre: claims.notaQueCompre,
    claimerUserId: claims.claimerUserId, claimerGuestId: claims.claimerGuestId,
    userNombre: users.nombre, guestNombre: guestIdentities.nombre,
  }).from(claims)
    .leftJoin(users, eq(claims.claimerUserId, users.id))
    .leftJoin(guestIdentities, eq(claims.claimerGuestId, guestIdentities.id))
    .where(and(inArray(claims.itemId, itemIds), inArray(claims.estado, ['reservado', 'comprado'])))

  const map = new Map<string, ClaimView[]>()
  for (const r of rows) {
    const view: ClaimView = {
      id: r.id,
      estado: r.estado as 'reservado' | 'comprado',
      claimerNombre: r.userNombre ?? r.guestNombre ?? 'Alguien',
      esMia: (!!viewer.userId && r.claimerUserId === viewer.userId) ||
             (!!viewer.guestId && r.claimerGuestId === viewer.guestId),
      notaQueCompre: r.notaQueCompre,
    }
    map.set(r.itemId, [...(map.get(r.itemId) ?? []), view])
  }
  return map
}

function toGiverView(row: ReturnType<typeof pickItemFields>, itemClaims: ClaimView[]): GiverItemView {
  const ocupados = itemClaims.length
  const quedan = Math.max(0, row.cantidad - ocupados)
  return { ...row, claims: itemClaims, quedan, disponible: row.repetible || quedan > 0 }
}

export async function getListForViewer(db: Db, shareToken: string, viewer: Viewer): Promise<ListView | null> {
  const [row] = await db.select({ list: lists, ownerNombre: users.nombre })
    .from(lists).innerJoin(users, eq(lists.ownerId, users.id))
    .where(eq(lists.shareToken, shareToken))
  if (!row) return null

  const allItems = await db.select().from(items).where(eq(items.listId, row.list.id))

  if (viewer.userId === row.list.ownerId) {
    return { role: 'owner', shareToken, items: allItems.map(pickItemFields) }
  }

  await expireStaleClaims(db)
  const activos = allItems.filter((i) => i.estadoDueno === 'activo')
  const claimMap = await loadClaimViews(db, activos.map((i) => i.id), viewer)
  return {
    role: 'giver', shareToken, ownerNombre: row.ownerNombre,
    items: activos.map((i) => toGiverView(pickItemFields(i), claimMap.get(i.id) ?? [])),
  }
}

export async function getOwnList(db: Db, userId: string) {
  const [list] = await db.select().from(lists).where(eq(lists.ownerId, userId))
  if (!list) return null
  const rows = await db.select().from(items).where(eq(items.listId, list.id))
  return { shareToken: list.shareToken, items: rows.map(pickItemFields) }
}

export type ItemDetail =
  | { role: 'owner'; item: OwnerItemView }
  | { role: 'giver'; ownerNombre: string; item: GiverItemView; comments: CommentView[] }

export async function getItemForViewer(db: Db, shareToken: string, itemId: string, viewer: Viewer): Promise<ItemDetail | null> {
  const [row] = await db.select({ item: items, ownerId: lists.ownerId, ownerNombre: users.nombre })
    .from(items)
    .innerJoin(lists, eq(items.listId, lists.id))
    .innerJoin(users, eq(lists.ownerId, users.id))
    .where(and(eq(items.id, itemId), eq(lists.shareToken, shareToken)))
  if (!row) return null

  if (viewer.userId === row.ownerId) {
    return { role: 'owner', item: pickItemFields(row.item) }
  }

  await expireStaleClaims(db, itemId)
  const claimMap = await loadClaimViews(db, [itemId], viewer)
  const commentRows = await db.select({
    id: comments.id, texto: comments.texto, createdAt: comments.createdAt,
    userNombre: users.nombre, guestNombre: guestIdentities.nombre,
  }).from(comments)
    .leftJoin(users, eq(comments.authorUserId, users.id))
    .leftJoin(guestIdentities, eq(comments.authorGuestId, guestIdentities.id))
    .where(eq(comments.itemId, itemId))
    .orderBy(comments.createdAt)

  return {
    role: 'giver', ownerNombre: row.ownerNombre,
    item: toGiverView(pickItemFields(row.item), claimMap.get(itemId) ?? []),
    comments: commentRows.map((c) => ({
      id: c.id, texto: c.texto, createdAt: c.createdAt,
      autorNombre: c.userNombre ?? c.guestNombre ?? 'Alguien',
    })),
  }
}
```

```ts
import { count, inArray, or, sql } from 'drizzle-orm' // sumar a los imports existentes
import { items, lists } from '@/db/schema'
import type { Viewer } from '@/lib/types'

export const RESERVA_DIAS = 14

export type ReserveResult =
  | { ok: true; claimId: string }
  | { ok: false; reason: 'propio' | 'sin_stock' | 'archivado' | 'no_existe' | 'sin_identidad' }

export async function reserveItem(db: Db, itemId: string, viewer: Viewer): Promise<ReserveResult> {
  if (!viewer.userId && !viewer.guestId) return { ok: false, reason: 'sin_identidad' }
  return await db.transaction(async (tx): Promise<ReserveResult> => {
    // lock del item: serializa reservas concurrentes sobre el mismo item
    const [row] = await tx.select({ item: items, ownerId: lists.ownerId })
      .from(items).innerJoin(lists, eq(items.listId, lists.id))
      .where(eq(items.id, itemId))
      .for('update', { of: items })
    if (!row) return { ok: false, reason: 'no_existe' }
    if (viewer.userId && viewer.userId === row.ownerId) return { ok: false, reason: 'propio' }
    if (row.item.estadoDueno !== 'activo') return { ok: false, reason: 'archivado' }

    await expireStaleClaims(tx as unknown as Db, itemId)

    if (!row.item.repetible) {
      const [{ n }] = await tx.select({ n: count() }).from(claims)
        .where(and(eq(claims.itemId, itemId), inArray(claims.estado, ['reservado', 'comprado'])))
      if (n >= row.item.cantidad) return { ok: false, reason: 'sin_stock' }
    }

    const [claim] = await tx.insert(claims).values({
      itemId,
      claimerUserId: viewer.userId ?? null,
      claimerGuestId: viewer.guestId ?? null,
      expiresAt: new Date(Date.now() + RESERVA_DIAS * 86400_000),
    }).returning()
    return { ok: true, claimId: claim.id }
  })
}

function claimEsDe(viewer: Viewer) {
  return viewer.userId
    ? eq(claims.claimerUserId, viewer.userId)
    : viewer.guestId
      ? eq(claims.claimerGuestId, viewer.guestId)
      : sql`false`
}

export async function markBought(db: Db, claimId: string, viewer: Viewer, notaQueCompre?: string): Promise<boolean> {
  const updated = await db.update(claims)
    .set({ estado: 'comprado', notaQueCompre: notaQueCompre ?? null, updatedAt: new Date() })
    .where(and(eq(claims.id, claimId), eq(claims.estado, 'reservado'), claimEsDe(viewer)))
    .returning({ id: claims.id })
  return updated.length > 0
}

export async function releaseClaim(db: Db, claimId: string, viewer: Viewer): Promise<boolean> {
  const updated = await db.update(claims)
    .set({ estado: 'liberado', updatedAt: new Date() })
    .where(and(eq(claims.id, claimId), eq(claims.estado, 'reservado'), claimEsDe(viewer)))
    .returning({ id: claims.id })
  return updated.length > 0
}

export async function getMyClaimWarnings(db: Db, viewer: Viewer) {
  if (!viewer.userId && !viewer.guestId) return []
  await expireStaleClaims(db)
  const rows = await db.select({
    claimId: claims.id, itemTitulo: items.titulo,
    createdAt: claims.createdAt, expiresAt: claims.expiresAt,
  }).from(claims).innerJoin(items, eq(claims.itemId, items.id))
    .where(and(eq(claims.estado, 'reservado'), claimEsDe(viewer)))
  const ahora = Date.now()
  return rows
    .filter((r) => r.expiresAt && ahora > (r.createdAt.getTime() + r.expiresAt.getTime()) / 2)
    .map((r) => ({
      claimId: r.claimId,
      itemTitulo: r.itemTitulo,
      diasRestantes: Math.ceil((r.expiresAt!.getTime() - ahora) / 86400_000),
    }))
}
```

```ts
  tipo: 'exacto' | 'idea'
  titulo: string
  notas?: string
  fotoUrl?: string
  precio?: string
  rangoPrecio?: string

  prioridad?: 'mucho' | 'gustaria' | 'suelta'

  const [list] = await db.select().from(lists).where(eq(lists.ownerId, userId))
  if (!list) return null

async function itemDelDueno(db: Db, itemId: string, userId: string) {
  const [row] = await db.select({ id: items.id })
    .from(items).innerJoin(lists, eq(items.listId, lists.id))
    .where(and(eq(items.id, itemId), eq(lists.ownerId, userId)))
  return !!row

async function notifyActiveClaimers(db: Db, itemId: string, tipo: 'item_baja' | 'item_editado') {
  const activos = await db.select().from(claims)
    .where(and(eq(claims.itemId, itemId), inArray(claims.estado, ['reservado', 'comprado'])))
  if (activos.length) {
    await db.insert(notifications).values(activos.map((c) => ({
      userId: c.claimerUserId, guestId: c.claimerGuestId, tipo, itemId,
    })))
  }
}

  await db.update(items).set(data).where(eq(items.id, itemId))
  await notifyActiveClaimers(db, itemId, 'item_editado')
  return true
}
```

```ts
export async function promoteGuest(db: Db, email: string, userId: string) {
  const guests = await db.select().from(guestIdentities)
    .where(and(eq(guestIdentities.email, email), isNull(guestIdentities.promotedToUserId)))
  for (const g of guests) {
    await db.update(claims).set({ claimerUserId: userId, claimerGuestId: null }).where(eq(claims.claimerGuestId, g.id))
    await db.update(comments).set({ authorUserId: userId, authorGuestId: null }).where(eq(comments.authorGuestId, g.id))
    await db.update(notifications).set({ userId, guestId: null }).where(eq(notifications.guestId, g.id))
    await db.update(guestIdentities).set({ promotedToUserId: userId }).where(eq(guestIdentities.id, g.id))
  }
}

// ponytail: promoción secuencial sin transacción; envolver en db.transaction si algún día importa la atomicidad parcial
```

```ts
import { randomBytes } from 'node:crypto'
import { lists, users } from '@/db/schema'
import type { Db } from '@/db/types'
import { promoteGuest } from '@/lib/guest'

export function newShareToken(): string {
  return randomBytes(8).toString('base64url')
}

export async function upsertUserFromGoogle(
  db: Db,
  p: { googleId: string; email: string; nombre: string; avatarUrl?: string | null },
) {
  const [user] = await db.insert(users)
    .values({ googleId: p.googleId, email: p.email, nombre: p.nombre, avatarUrl: p.avatarUrl ?? null })
    .onConflictDoUpdate({ target: users.googleId, set: { nombre: p.nombre, avatarUrl: p.avatarUrl ?? null } })
    .returning()
  await db.insert(lists)
    .values({ ownerId: user.id, shareToken: newShareToken() })
    .onConflictDoNothing({ target: lists.ownerId })
  await promoteGuest(db, p.email, user.id)
  return user
}
```

```ts
  const session = await auth()
```

```ts
async function viewerIdentificandoGuest(formData: FormData): Promise<Viewer> {
```

```tsx
import Link from 'next/link'
```

```tsx
import { db } from '@/db'
import { getViewer } from '@/lib/viewer'
import { getUnreadNotifications } from '@/lib/notifications'
import { getMyClaimWarnings } from '@/lib/claims'
import { markNotifsReadAction } from '@/app/actions'

const COPY = {
  item_baja: (t: string) => `“${t}” fue dado de baja por su dueño — te conviene elegir otro regalo.`,
  item_editado: (t: string) => `“${t}” fue editado por su dueño — revisá los detalles.`,
} as const

export async function Banners() {
  const viewer = await getViewer()
  const [notifs, warnings] = await Promise.all([
    getUnreadNotifications(db, viewer),
    getMyClaimWarnings(db, viewer),
  ])
  if (!notifs.length && !warnings.length) return null

  return (
    <div className="mt-4 space-y-2">
      {notifs.map((n) => (
        <div key={n.id} className="rounded-2xl border border-amber-300 bg-amber-50 p-3 text-sm">
          {COPY[n.tipo](n.itemTitulo)}
        </div>
      ))}
      {warnings.map((w) => (
        <div key={w.claimId} className="rounded-2xl border border-red-300 bg-red-50 p-3 text-sm">
          ¿Conseguiste “{w.itemTitulo}”? Te quedan {w.diasRestantes} {w.diasRestantes === 1 ? 'día' : 'días'} de reserva.
        </div>
      ))}
      {notifs.length > 0 && (
        <form action={markNotifsReadAction}>
          <button className="text-xs text-stone-400 underline">Marcar como leídas</button>
        </form>
      )}
    </div>
  )
}
```
